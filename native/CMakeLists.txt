option(FFR_BUILD_NATIVE_OBSERVER "Build the exact-profile experimental Windows observer DLL" OFF)
option(FFR_BUILD_ABORT_TRIAL "Build the separate, disabled-by-default opening-abort trial DLL" OFF)
if(FFR_BUILD_ABORT_TRIAL AND NOT FFR_BUILD_NATIVE_OBSERVER)
  message(FATAL_ERROR "The abort trial requires the native observer build and verified dependency")
endif()
if(FFR_BUILD_NATIVE_OBSERVER)
  if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The native observer requires a Windows x64 game environment")
  endif()
  if(NOT EXISTS "${MINHOOK_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Pass MINHOOK_SOURCE_DIR for the verified MinHook 1.3.4 source")
  endif()
  enable_language(C)
  add_subdirectory("${MINHOOK_SOURCE_DIR}" minhook EXCLUDE_FROM_ALL)
  add_executable(load_route_hook_tests tests/load_route_hook_tests.cpp)
  target_include_directories(load_route_hook_tests PRIVATE include)
  target_link_libraries(load_route_hook_tests PRIVATE minhook Threads::Threads)
  target_compile_options(load_route_hook_tests PRIVATE /W4 /EHsc /utf-8)
  add_test(NAME isolated_native_load_route_detours COMMAND load_route_hook_tests)
  add_executable(native_load_binding_tests tests/native_load_binding_tests.cpp)
  target_include_directories(native_load_binding_tests PRIVATE include)
  target_link_libraries(native_load_binding_tests PRIVATE minhook bcrypt Threads::Threads)
  target_compile_options(native_load_binding_tests PRIVATE /W4 /EHsc /utf-8)
  foreach(mode good build-mismatch profile-mismatch callsite-mismatch dependency-mismatch dependency-changed wrong-thread wrong-context wrong-class post-context)
    add_test(NAME native_load_binding_${mode} COMMAND native_load_binding_tests "${mode}")
  endforeach()
  add_library(FFRHelperRuntimeNative SHARED src/native_observer.cpp)
  target_include_directories(FFRHelperRuntimeNative PRIVATE include)
  target_link_libraries(FFRHelperRuntimeNative PRIVATE minhook bcrypt)
  target_compile_definitions(FFRHelperRuntimeNative PRIVATE UNICODE _UNICODE)
  target_compile_options(FFRHelperRuntimeNative PRIVATE /W4 /EHsc /utf-8)
  set_property(TARGET FFRHelperRuntimeNative PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
  set_target_properties(FFRHelperRuntimeNative PROPERTIES PREFIX "" OUTPUT_NAME "main")
  if(FFR_BUILD_ABORT_TRIAL)
    add_library(FFRHelperBattlePractice SHARED src/native_observer.cpp)
    target_include_directories(FFRHelperBattlePractice PRIVATE include)
    target_link_libraries(FFRHelperBattlePractice PRIVATE minhook bcrypt)
    target_compile_definitions(FFRHelperBattlePractice PRIVATE UNICODE _UNICODE FFR_PRACTICE_BUILD=1)
    target_compile_options(FFRHelperBattlePractice PRIVATE /W4 /EHsc /utf-8)
    set_property(TARGET FFRHelperBattlePractice PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
  endif()

  # Fixture modules use generated mock-only fingerprints and different filenames.
  # They are never packaged with the production observer.
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_library(native_mock_loader SHARED tests/mock_loader.cpp)
  set_target_properties(native_mock_loader PROPERTIES OUTPUT_NAME "UE4SS")
  add_executable(native_host_fixture tests/native_host_fixture.cpp)
  target_include_directories(native_host_fixture PRIVATE include)
  target_compile_options(native_host_fixture PRIVATE /W4 /EHsc /utf-8)
  set(generated "${CMAKE_CURRENT_BINARY_DIR}/fixture_profiles")
  set(profiles "${generated}/good_profile.hpp" "${generated}/bad_game_profile.hpp" "${generated}/bad_loader_profile.hpp" "${generated}/bad_function_profile.hpp" "${generated}/good_abort_profile.hpp" "${generated}/good_import_profile.hpp" "${generated}/good_field_profile.hpp" "${generated}/good_playtime_profile.hpp")
  add_custom_command(OUTPUT ${profiles}
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/make_fixture_profiles.py"
      "$<TARGET_FILE:native_host_fixture>" "$<TARGET_FILE:native_mock_loader>" "${generated}"
    DEPENDS native_host_fixture native_mock_loader tests/make_fixture_profiles.py
    VERBATIM)
  add_custom_target(generate_fixture_profiles DEPENDS ${profiles})
  foreach(mode good bad_game bad_loader bad_function)
    add_library(fixture_observer_${mode} SHARED src/native_observer.cpp)
    add_dependencies(fixture_observer_${mode} generate_fixture_profiles)
    target_include_directories(fixture_observer_${mode} PRIVATE include "${generated}")
    target_link_libraries(fixture_observer_${mode} PRIVATE minhook bcrypt)
    target_compile_options(fixture_observer_${mode} PRIVATE /W4 /EHsc /utf-8)
    target_compile_definitions(fixture_observer_${mode} PRIVATE UNICODE _UNICODE "FFR_CAPTURE_PROFILE_HEADER=\"${mode}_profile.hpp\"" "FFR_FIELD_PROFILE_HEADER=\"good_field_profile.hpp\"" "FFR_PLAYTIME_PROFILE_HEADER=\"good_playtime_profile.hpp\"")
    set_property(TARGET fixture_observer_${mode} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
    set_target_properties(fixture_observer_${mode} PROPERTIES OUTPUT_NAME "fixture_observer_${mode}")
    string(REPLACE "_" "-" case "${mode}")
    add_test(NAME native_observer_${mode}
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_native_fixture.py"
        --exe "$<TARGET_FILE:native_host_fixture>" --loader "$<TARGET_FILE:native_mock_loader>"
        --module "$<TARGET_FILE:fixture_observer_${mode}>" --mode "${case}" --work "${CMAKE_CURRENT_BINARY_DIR}/test-work")
  endforeach()
  if(FFR_BUILD_ABORT_TRIAL)
    add_library(fixture_observer_abort SHARED src/native_observer.cpp)
    add_dependencies(fixture_observer_abort generate_fixture_profiles)
    target_include_directories(fixture_observer_abort PRIVATE include "${generated}")
    target_link_libraries(fixture_observer_abort PRIVATE minhook bcrypt)
    target_compile_options(fixture_observer_abort PRIVATE /W4 /EHsc /utf-8)
    target_compile_definitions(fixture_observer_abort PRIVATE UNICODE _UNICODE FFR_PRACTICE_BUILD=1
      "FFR_CAPTURE_PROFILE_HEADER=\"good_profile.hpp\"" "FFR_ABORT_PROFILE_HEADER=\"good_abort_profile.hpp\"" "FFR_IMPORT_PROFILE_HEADER=\"good_import_profile.hpp\"" "FFR_FIELD_PROFILE_HEADER=\"good_field_profile.hpp\"" "FFR_PLAYTIME_PROFILE_HEADER=\"good_playtime_profile.hpp\"")
    foreach(mode abort abort-denied clone clone-unready clone-mismatch clone-mutates clone-denied combat combat-after-action combat-denied combat-stale recovery recovery-denied recovery-stale recovery-drift recovery-write-failed recovery-map-changed)
      add_test(NAME native_${mode} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_native_fixture.py"
        --exe "$<TARGET_FILE:native_host_fixture>" --loader "$<TARGET_FILE:native_mock_loader>"
        --module "$<TARGET_FILE:fixture_observer_abort>" --mode "${mode}" --work "${CMAKE_CURRENT_BINARY_DIR}/test-work")
    endforeach()
  endif()
endif()
